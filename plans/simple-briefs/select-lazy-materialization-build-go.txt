## The design that already worked (mirror this; you're rebuilding it from a clean checkout, not
## resuming a branch)

A `select` becomes a lazy handle: build a selection vector (survivor indices) once, on first use;
answer indexing/length from that vector in O(1) without copying survivors out; only a strong
consumer (something that needs a real `[]T`) forces a full materialization ("dense").

```go
type tlSel[T any] struct {
	src     []T
	pred    func(T) bool
	idx     []int32
	dense   []T
	inPlace bool
}

func tlSelNew[T any](src []T, pred func(T) bool) *tlSel[T] {
	return &tlSel[T]{src: src, pred: pred}
}

func (s *tlSel[T]) build() {
	if s.idx != nil {
		return
	}
	idx := make([]int32, 0, len(s.src))
	for i, e := range s.src {
		if s.pred(e) {
			idx = append(idx, int32(i))
		}
	}
	s.idx = idx
}

func tlSelAt[T any](s *tlSel[T], i int32) tlOpt[T] {
	s.build()
	n := int32(len(s.idx))
	if i < 0 {
		i = n + i
	}
	if i < 0 || i >= n {
		return tlOpt[T]{}
	}
	return tlOpt[T]{true, s.src[s.idx[i]]}
}

func tlSelLen[T any](s *tlSel[T]) int32 {
	s.build()
	return int32(len(s.idx))
}

func tlSelDense[T any](s *tlSel[T]) []T {
	if s.dense != nil {
		return s.dense
	}
	s.build()
	out := make([]T, len(s.idx))
	for j, i := range s.idx {
		out[j] = s.src[i]
	}
	s.dense = out
	return out
}
```

Wire every other consumer (slice, map, print, fusion, iteration) to call `tlSelDense` when it
needs a real `[]T` -- that is always correct, just not lazy for that path. The `inPlace`/reuse
variant (reusing the source's backing array when the source has exactly one use, per
`mutation-semantics-spike`'s provably-one-reference condition) is a perf nicety, not a
correctness requirement for this row's done-gate -- skip it if you're short on time; a plain
`tlSelNew` that always copies on `tlSelDense` is a correct, acceptable implementation.
