flattened list:

```toylang
fn reverse_num(p: {n: Int, acc: Int}) -> Int =
    p
        | .n == 0 -> p.acc or
              reverse_num({n: p.n / 10, acc: p.acc * 10 + p.n % 10})

fn row_candidates(a: Int) -> Vec<Int> =
    collect(range(1000))
        | select(. >= a)
        | map(a * .)
        | select(. == reverse_num({n: ., acc: 0}))

max(
    flatten(
        collect(range(1000)) | select(. >= 100) | map(row_candidates(.))
    )
)!
```

```toylang
fn window(p: {v: Vec<Int>, i: Int, k: Int}) -> Int64 =
    p
        | .k == 0 -> 1 or
              i64(p.v[p.i + p.k - 1]!) * window({v: p.v, i: p.i, k: p.k - 1})

fn windows(v: Vec<Int>) -> Vec<Int64> =
    collect(range(length(v) - 12)) | map(window({v: v, i: ., k: 13}))

max(windows(parse(stdin)))!
```
